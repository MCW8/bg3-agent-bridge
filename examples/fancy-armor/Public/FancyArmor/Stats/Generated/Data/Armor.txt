// Fancy Armor — looks like plate, counts as light.
//
// ArmorType is what drives proficiency, not the appearance:
//   Light   Padded, Leather, StuddedLeather
//   Medium  Hide, ChainShirt, ScaleMail, BreastPlate, HalfPlate
//   Heavy   RingMail, ChainMail, Splint, Plate
//
// So inheriting ARM_StuddedLeather_Body gives light-armour behaviour — full
// Dexterity to AC, no Strength requirement, and none of the stealth
// disadvantage that ARM_Plate_Body carries via Boosts. The plate look comes
// entirely from the root template's ParentTemplateId.
//
// RootTemplate must point back at the template whose Stats names this entry.
new entry "ARM_FancyArmor"
type "Armor"
using "ARM_StuddedLeather_Body"
data "RootTemplate" "ff2c72c3-7301-4f6f-bcad-3f98711a0321"
data "Slot" "Breast"
data "ArmorType" "StuddedLeather"
data "ArmorClass" "12"
data "Boosts" ""
data "Rarity" "Rare"
data "ValueLevel" "6"
data "Weight" "6.0"
data "Unique" "1"
