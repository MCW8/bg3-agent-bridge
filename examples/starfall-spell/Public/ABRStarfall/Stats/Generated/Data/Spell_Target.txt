// Starfall — a ground-targeted radiant burst.
//
// Inherits Target_FaerieFire for its animation, cast effects and sounds. Faerie
// Fire is the better parent than Fireball here: it is already a Target spell
// rather than a Projectile, and its SpellProperties are empty, so nothing
// unwanted (such as Grease's surface) comes along with the inheritance.
//
// The two overrides that make it castable anywhere:
//   TargetConditions  "not Item();"  — copied from Target_Grease, which is the
//                                      permissive form that allows bare ground.
//   SpellFlags        drops IsConcentration, which Faerie Fire carries.
new entry "Target_ABR_Starfall"
type "SpellData"
data "SpellType" "Target"
using "Target_FaerieFire"
data "Level" "0"
data "SpellSchool" "Evocation"
data "AreaRadius" "6"
data "TargetRadius" "18"
data "TargetConditions" "not Item();"
data "SpellFlags" "HasVerbalComponent;IsSpell;IsHarmful;RangeIgnoreVerticalThreshold;HasHighGroundRangeExtension"
data "UseCosts" "ActionPoint:1"
data "SpellRoll" "not SavingThrow(Ability.Dexterity, SourceSpellDC())"
data "SpellSuccess" "ApplyStatus(FAERIE_FIRE,100,10);DealDamage(2d4,Radiant)"
data "SpellFail" ""
data "VerbalIntent" "Debuff"
data "Icon" "Spell_Evocation_FaerieFire"
data "DisplayName" "h855bc763g398eg4df2g8e4bg221b7f0cd0e1;1"
data "Description" "h28d03ec8gd097g458egbbe1gf8a3b78f0256;1"
data "TooltipStatusApply" "ApplyStatus(FAERIE_FIRE,100,10)"
